Cascade, héritage & spécificité

/* Le plus spécifique (ou le plus tardif) gagne */
p { color:#666; }          /* élément */
.article p { color:#333; } /* classe + élément */
#main p { color:#111; }    /* id + élément (le plus fort ici) */
