Calculatrice simple (addition/soustraction/multiplication/division)

const a = parseFloat(document.getElementById('calc-a').value) || 0;
const b = parseFloat(document.getElementById('calc-b').value) || 0;
const op = document.querySelector('input[name="calc-op"]:checked').value;
let res = NaN;
switch (op) {
  case 'add': res = a + b; break;
  case 'sub': res = a - b; break;
  case 'mul': res = a * b; break;
  case 'div': res = b !== 0 ? a / b : NaN; break;
}
document.getElementById('calc-res').textContent = isNaN(res) ? 'Erreur' : res;

Mini convertisseur °C ⇄ °F


  const $ = (s, r=document) => r.querySelector(s);

  $('#toF')?.addEventListener('click', () => {
    const c = parseFloat($('#celsius').value);
    const f = isNaN(c) ? '—' : (c * 9/5 + 32).toFixed(2);
    $('#outF').textContent = f;
  });

  $('#toC')?.addEventListener('click', () => {
    const f = parseFloat($('#fahrenheit').value);
    const c = isNaN(f) ? '—' : ((f - 32) * 5/9).toFixed(2);
    $('#outC').textContent = c;
  });

  $('#celsius')?.addEventListener('keydown', (e) => {
    if (e.key === 'Enter') $('#toF').click();
  });
  $('#fahrenheit')?.addEventListener('keydown', (e) => {
    if (e.key === 'Enter') $('#toC').click();
  });

Délégation d’événements + Templates HTML

<template id="todo-tpl">
  <li class="todo">
    <span class="txt"></span>
    <button class="btn secondary" data-action="remove">Supprimer</button>
  </li>
</template>
const tpl = document.getElementById('todo-tpl');
function addTodo(text){
  const node = tpl.content.cloneNode(true);
  node.querySelector('.txt').textContent = text;
  document.getElementById('todo-list').append(node);
}
document.addEventListener('click', (e) => {
  const rm = e.target.closest('[data-action="remove"]');
  if (!rm) return;
  e.preventDefault();
  rm.closest('li')?.remove();
});

      Performance : debounce (saisie) & throttle (scroll)

      const debounce = (fn, d=300) => { let t; return (...a) => { clearTimeout(t); t=setTimeout(()=>fn(...a), d); }; };
const throttle = (fn, w=150) => { let p=0; return (...a) => { const n=Date.now(); if (n-p>=w){ p=n; fn(...a); } }; };
